    /* --- LOAD AND RESOLVE PAGE --- */
    async function resolvePage(pageNum, timeoutMs = config.pageTimeout) {
        const selector = config.selectors.pageContainer(pageNum);
        const outerSelector = config.selectors.outerContainer(pageNum);
        let pageEl = document.querySelector(selector);
        let outerEl = document.querySelector(outerSelector);
        const scroller = document.querySelector('.document_scroller') || document.documentElement || window;

        const start = Date.now();
        let scrollAttempt = 0;

        while (Date.now() - start < timeoutMs) {
            if (!isRunning) return null;

            if (!outerEl) outerEl = document.querySelector(outerSelector);
            if (!pageEl) pageEl = document.querySelector(selector);

            // Wake up Scribd's lazy loader if available
            if (window.docManager) {
                try {
                    if (typeof window.docManager.lazyLoad === 'function') window.docManager.lazyLoad();
                    if (typeof window.docManager.scrollToPage === 'function') window.docManager.scrollToPage(pageNum);
                    if (typeof window.docManager.gotoPage === 'function') window.docManager.gotoPage(pageNum);
                } catch(e) {}
            }

            if (!outerEl) {
                if (scroller.scrollBy) scroller.scrollBy(0, config.scrollStep);
                else window.scrollBy(0, config.scrollStep);

                if (scroller.dispatchEvent) scroller.dispatchEvent(new Event('scroll'));
                window.dispatchEvent(new Event('scroll'));
                await wait(150);
                continue;
            }

            if (!pageEl) {
                outerEl.scrollIntoView({ block: "center", behavior: "auto" });
                if (scroller.dispatchEvent) scroller.dispatchEvent(new Event('scroll'));
                window.dispatchEvent(new Event('scroll'));
                document.dispatchEvent(new Event('scroll'));
                await wait(200);

                pageEl = document.querySelector(selector);
                if (!pageEl) {
                    scrollAttempt++;
                    const step = scrollAttempt % 2 === 0 ? 80 : -80;
                    if (scroller.scrollBy) scroller.scrollBy(0, step);
                    else window.scrollBy(0, step);

                    if (scroller.dispatchEvent) scroller.dispatchEvent(new Event('scroll'));
                    window.dispatchEvent(new Event('scroll'));
                    await wait(200);
                    continue;
                }
            }

            if (outerEl.classList.contains('blurred_page')) {
                outerEl.classList.remove('blurred_page');
            }
            const textLayer = pageEl.querySelector(config.selectors.textLayer);
            if (textLayer) {
                textLayer.style.textShadow = 'none';
                textLayer.style.color = '#000000';
            }

            const imgs = Array.from(pageEl.querySelectorAll(config.selectors.image));
            imgs.forEach(img => {
                const orig = img.getAttribute('orig') || img.getAttribute('data-src');
                if (orig && (!img.src || img.src.includes('loading') || img.src.startsWith('data:'))) {
                    img.src = orig.replace(/^http:\/\/html\.scribd\.com/, 'https://html.scribdassets.com');
                }
            });

            const allImgsLoaded = imgs.length === 0 || imgs.every(img => img.complete && img.naturalWidth > 0);
            const hasText = textLayer && (textLayer.innerText || textLayer.textContent || '').trim().length > 0;

            if (hasText && document.fonts && document.fonts.ready) {
                await document.fonts.ready;
            }

            if (allImgsLoaded && (imgs.length > 0 || hasText || pageEl.children.length > 0)) {
                return pageEl;
            }

            await wait(config.pollInterval);
        }

        return pageEl || null;
    }

    /* --- CAPTURE HIGH-RES IMAGE WITH EXACT PIXEL DIMENSIONS --- */
    async function fetchImageObject(pageEl, pageNum) {
        const imgs = Array.from(pageEl.querySelectorAll(config.selectors.image));
        let bestImg = null;
        let src = '';
        let bestWidth = 0;

        for (const img of imgs) {
            const orig = img.getAttribute('orig') || img.getAttribute('data-src') || img.src || '';
            const style = img.getAttribute('style') || '';
            const wMatch = style.match(/width:\s*(\d+)px/);
            const w = wMatch ? parseInt(wMatch[1]) : (img.naturalWidth || img.offsetWidth || 0);

            if (orig.includes('/pages/') || orig.includes('/images/') || orig.includes('.jpg') || orig.includes('.jpeg') || w > bestWidth) {
                if (w >= bestWidth || orig.includes('/pages/')) {
                    bestWidth = w;
                    bestImg = img;
                    src = orig;
                }
            }
        }

        if (!src && imgs.length > 0) {
            bestImg = imgs[0];
            src = bestImg.getAttribute('orig') || bestImg.getAttribute('data-src') || bestImg.src || '';
        }

        if (!src) {
            const bgDiv = pageEl.querySelector('.image_layer, [style*="background-image"]');
            if (bgDiv) {
                const match = (bgDiv.getAttribute('style') || '').match(/url\(["']?([^"']+)["']?\)/);
                if (match) src = match[1];
            }
        }

        if (!src) return null;

        src = src.replace(/^http:\/\/html\.scribd\.com/, 'https://html.scribdassets.com');

        let dataUrl = '';
        let width = 0;
        let height = 0;

        try {
            const res = await fetch(src);
            const blob = await res.blob();
            dataUrl = await new Promise((resolve) => {
                const reader = new FileReader();
                reader.onloadend = () => resolve(reader.result);
                reader.readAsDataURL(blob);
            });

            await new Promise((resolve) => {
                const testImg = new Image();
                testImg.onload = () => {
                    width = testImg.naturalWidth;
                    height = testImg.naturalHeight;
                    resolve();
                };
                testImg.onerror = resolve;
                testImg.src = dataUrl;
            });
        } catch (e) {
            console.warn("Direct blob load fallback:", e);
            if (bestImg && bestImg.naturalWidth > 0) {
                width = bestImg.naturalWidth;
                height = bestImg.naturalHeight;
                dataUrl = bestImg.src;
            }
        }

        if (!width && bestImg && bestImg.naturalWidth > 0) {
            width = bestImg.naturalWidth;
            height = bestImg.naturalHeight;
        }

        if (!width || !height) {
            const style = (bestImg && bestImg.getAttribute('style')) || (pageEl.getAttribute('style')) || '';
            const wMatch = style.match(/width:\s*(\d+)px/);
            const hMatch = style.match(/height:\s*(\d+)px/);
            width = wMatch ? parseInt(wMatch[1]) : (pageEl.offsetWidth || 900);
            height = hMatch ? parseInt(hMatch[1]) : (pageEl.offsetHeight || 1200);
        }

        return {
            dataUrl: dataUrl || src,
            width: width,
            height: height,
            src: src,
            pageNum: pageNum
        };
    }

    /* --- BUILD NATIVE VARIABLE-DIMENSION & LANDSCAPE/PORTRAIT PDF --- */
    async function buildDirectPDF() {
        if (collectedImages.length === 0) {
            alert("Hiçbir sayfa görseli toplanamadı!");
            return;
        }

        log(`PDF oluşturuluyor (${collectedImages.length} sayfa)...`, "#9b59b6");
        const { jsPDF } = window.jspdf;

        const first = collectedImages[0];
        const w0 = first.width;
        const h0 = first.height;
        const orient0 = w0 > h0 ? 'landscape' : 'portrait';

        const pdf = new jsPDF({
            orientation: orient0,
            unit: 'px',
            format: [w0, h0],
            hotfixes: ['px_scaling']
        });

        for (let i = 0; i < collectedImages.length; i++) {
            const p = collectedImages[i];
            const w = p.width;
            const h = p.height;
            const orient = w > h ? 'landscape' : 'portrait';

            if (i > 0) {
                pdf.addPage([w, h], orient);
            }

            pdf.addImage(p.dataUrl, 'JPEG', 0, 0, w, h, undefined, 'FAST');
            log(`PDF'e Sayfa ${i + 1}/${collectedImages.length} ekleniyor (${orient} ${w}x${h}px)...`, "#9b59b6");
            await wait(10);
        }

        log("PDF kaydediliyor...", "#27ae60");
        pdf.save(`${bookName}.pdf`);
        document.title = originalTitle;
        log("İndirme Tamamlandı!", "#27ae60");
        alert("PDF orijinal sayfa boyutları ve yönlendirmeleriyle (yatay/dikey tam sayfa) başarıyla indirildi!");
    }

    /* --- DIRECT IMAGE SAVER --- */
    async function saveSingleImage(captured, pageNum) {
        try {
            const res = await fetch(captured.src);
            const blob = await res.blob();
            const url = URL.createObjectURL(blob);
            const a = document.createElement('a');
            a.href = url;
            a.download = `${bookName}_Page_${String(pageNum).padStart(3, '0')}.jpg`;
            document.body.appendChild(a);
            a.click();
            setTimeout(() => {
                document.body.removeChild(a);
                URL.revokeObjectURL(url);
            }, 1000);
        } catch (e) {
            console.error("Direct image download error:", e);
        }
    }

    /* --- TRIGGER CLEAN PRINT MODAL (ACCURATE TITLE & NO PANEL) --- */
    async function triggerCleanPrint() {
        ensurePrintStyles();

        // Set document.title to clean bookName so the browser uses it as default PDF filename!
        document.title = bookName;

        // Completely detach panel from DOM so print preview is 100% clean
        const parent = panel.parentNode;
        const nextSib = panel.nextSibling;
        panel.remove();

        const restore = () => {
            if (parent && !document.getElementById(panelId)) {
                parent.insertBefore(panel, nextSib);
            }
            document.title = originalTitle;
        };

        window.addEventListener('afterprint', restore, { once: true });
        window.addEventListener('focus', restore, { once: true });

        await wait(250);
        window.print();
    }

    /* --- MAIN ACQUISITION LOOP --- */
    async function startLoop() {
        while (isRunning) {
            if (pageCounter > maxPages) {
                isRunning = false;
                progressBar.style.width = "100%";
                progressPercent.innerText = "100%";

                if (downloadMode === 'jspdf') {
                    await buildDirectPDF();
                    goBtn.innerText = "TAMAMLANDI";
                    goBtn.style.background = "#27ae60";
                } else if (downloadMode === 'print') {
                    log("Tüm sayfalar yüklendi! Vektörel Baskı açılıyor...", "#27ae60");
                    ensurePrintStyles();

                    for (let p = 1; p <= initialMax; p++) {
                        const outer = document.querySelector(config.selectors.outerContainer(p));
                        if (outer) {
                            if (p < startPageNum || p > maxPages) {
                                outer.classList.add('sd-print-excluded');
                            } else {
                                outer.classList.remove('sd-print-excluded');
                            }
                        }
                    }

                    window.scrollTo(0, 0);
                    await wait(300);

                    goBtn.innerText = "BASKI PENCERESİNİ AÇ";
                    goBtn.style.background = "#27ae60";

                    await triggerCleanPrint();
                } else {
                    log("Tüm Görseller İndirildi!", "#27ae60");
                    alert("Görseller başarıyla indirildi!");
                    goBtn.innerText = "TAMAMLANDI";
                    goBtn.style.background = "#27ae60";
                }
                break;
            }

            updateProgress();
            log(`Sayfa ${pageCounter} işleniyor...`, "#fff");

            const pageEl = await resolvePage(pageCounter);

            if (!pageEl) {
                log(`Sayfa ${pageCounter} yüklenemedi. Durduruldu.`, "#e74c3c");
                isRunning = false;
                break;
            }

            // If on initial page, re-verify document type from rendered page
            if (pageCounter === startPageNum) {
                const verifiedType = detectDocumentType();
                if (verifiedType !== docType) {
                    docType = verifiedType;
                    isDocx = docType === 'docx';
                    docBadge.innerText = isDocx ? '📝 METİN / DOCX (Vektörel)' : '📚 RESİMLİ KİTAP (Tam Sayfa)';
                    docBadge.style.background = isDocx ? '#9b59b6' : '#27ae60';
                }
            }

            if (downloadMode === 'jspdf') {
                const imgObj = await fetchImageObject(pageEl, pageCounter);
                if (imgObj) {
                    collectedImages.push(imgObj);
                    log(`Sayfa ${pageCounter} alındı (${collectedImages.length} hazır)...`, "#27ae60");
                }
            } else if (downloadMode === 'images') {
                const imgObj = await fetchImageObject(pageEl, pageCounter);
                if (imgObj) {
                    log(`Sayfa ${pageCounter} JPG kaydediliyor...`, "#27ae60");
                    await saveSingleImage(imgObj, pageCounter);
                    await wait(250);
                }
            }

            pageCounter++;
            startInput.value = pageCounter;
            await wait(60);
        }
    }

    /* --- START / RESUME HANDLER --- */
    goBtn.onclick = async () => {
        if (goBtn.innerText === "BASKI PENCERESİNİ AÇ") {
            await triggerCleanPrint();
            return;
        }

        if (isRunning) {
            isRunning = false;
            goBtn.innerText = "DEVAM ET";
            goBtn.style.background = "#1abc9c";
            log("Duraklatıldı.", "#f39c12");
            return;
        }

        if (!isContentMode()) {
            if (confirm("İndirme işleminin paywall/bulanıklık engeline takılmaması için CONTENT MODE önerilir.\nContent Mode'a geçilsin mi?")) {
                goToContentMode();
                return;
            }
        }

        isRunning = true;
        pageCounter = parseInt(startInput.value);
        startPageNum = pageCounter;
        maxPages = parseInt(endInput.value);

        if (goBtn.innerText !== "DEVAM ET") {
            collectedImages = [];
        }

        goBtn.innerText = "DURDUR";
        goBtn.style.background = "#e74c3c";
        log("İndirme başlatılıyor...", "#fff");
        updateProgress();

        startLoop();
    };
})();
