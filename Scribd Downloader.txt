// ==UserScript==
// @name         Scribd Embed Downloader v2.1 (PDF + Progress)
// @version      2.1
// @description  Downloads pages from Scribd Embeds with PDF export and progress bar.
// @author       You
// @match        https://www.scribd.com/embeds/*/content*
// ==/UserScript==

javascript:(async function(){
    /* --- LOAD JSPDF LIBRARY --- */
    if (!window.jspdf) {
        const script = document.createElement('script');
        script.src = 'https://cdnjs.cloudflare.com/ajax/libs/jspdf/2.5.1/jspdf.umd.min.js';
        document.head.appendChild(script);
        await new Promise(resolve => script.onload = resolve);
    }

    /* --- CAPTURE TITLE FIRST --- */
    const originalTitle = document.title;

    /* --- CONFIGURATION --- */
    const config = {
        concurrency: 4,     // pages fetched in parallel
        pollInterval: 100,  // how often to check if a page image has loaded
        pageTimeout: 8000,  // give up on a page after this long
        scrollStep: 800,    // px to scroll when a page container is not rendered yet
        saveGap: 250,       // delay between file saves (Safari drops simultaneous downloads)
        retryAttempts: 3,
        retryDelay: 1000,
        selectors: {
            pageContainer: (num) => `div#page${num}`,
            image: 'img.absimg',
            pageCount: ['div[data-e2e="total-pages"]', '.pageCount'],
            titleLink: 'a[data-e2e="full-screen-embed"]'
        }
    };

    /* --- HELPERS --- */
    function getBookTitle() {
        try {
            const link = document.querySelector(config.selectors.titleLink);
            if (link && link.href) {
                const parts = link.href.split('/');
                let slug = parts[5] || parts[4] || "Scribd_Doc";
                slug = slug.split('#')[0].split('?')[0];
                return slug.replace(/[^a-zA-Z0-9\-_]/g, '_');
            }
        } catch(e) {}
        return originalTitle.replace(/[^a-zA-Z0-9\-_ ]/g, '').trim() || "Scribd_Doc";
    }

    function getTotalPages() {
        try {
            const selectors = Array.isArray(config.selectors.pageCount) 
                ? config.selectors.pageCount 
                : [config.selectors.pageCount];
            
            for (let selector of selectors) {
                const el = document.querySelector(selector);
                if (el && el.innerText) {
                    const txt = el.innerText.trim();
                    const num = parseInt(txt.replace('/', '').trim());
                    if (!isNaN(num) && num > 0) {
                        return num;
                    }
                }
            }
        } catch(e) {
            console.log("Error detecting total pages:", e);
        }
        return 9999;
    }

    /* --- URL HELPERS --- */
    function isContentMode() {
        return window.location.href.includes('/embeds/') && window.location.href.includes('/content');
    }

    function getDocumentId() {
        const url = window.location.href;
        const match = url.match(/(?:document|embeds)\/(\d+)/);
        return match ? match[1] : null;
    }

    function goToContentMode() {
        const docId = getDocumentId();
        if (docId) {
            const embedUrl = `https://www.scribd.com/embeds/${docId}/content`;
            window.location.href = embedUrl;
        } else {
            alert('Could not detect document ID from URL.');
        }
    }

    /* --- UI PANEL --- */
    const panelId = 'sd-panel-v2.0';
    if(document.getElementById(panelId)) document.getElementById(panelId).remove();

    const initialMax = getTotalPages();
    const bookName = getBookTitle();
    const inContentMode = isContentMode();

    const panel = document.createElement('div');
    panel.id = panelId;
    panel.style.cssText = "position:fixed;top:10px;right:10px;width:320px;background:#1e1e1e;color:#fff;padding:15px;border:2px solid #1abc9c;z-index:2147483647;font-family:sans-serif;box-shadow:0 0 20px rgba(0,0,0,0.8);font-size:12px;border-radius:8px;";

    panel.innerHTML = `
        <h3 style="margin:0 0 10px;text-align:center;border-bottom:1px solid #333;padding-bottom:8px;color:#1abc9c;font-weight:bold;">SCRIBD DOWNLOADER <span style="font-size:10px;">v2.1</span></h3>

        <button id="sd-content" ${inContentMode ? 'disabled' : ''} style="width:100%;background:${inContentMode ? '#27ae60' : '#e67e22'};color:#fff;border:none;padding:10px;cursor:${inContentMode ? 'not-allowed' : 'pointer'};font-weight:bold;font-size:12px;border-radius:4px;margin-bottom:10px;opacity:${inContentMode ? '0.7' : '1'};">
            ${inContentMode ? '✓ CONTENT MODE ACTIVE' : 'GO TO CONTENT MODE'}
        </button>

        <div style="margin-bottom:10px;background:#2c2c2c;padding:8px;border-radius:4px;">
            <div style="color:#aaa;font-size:10px;margin-bottom:5px;text-align:center;">DOWNLOAD MODE</div>
            <div style="display:flex;gap:5px;">
                <button id="sd-mode-images" style="flex:1;background:#3498db;color:#fff;border:none;padding:8px;cursor:pointer;font-weight:bold;border-radius:4px;font-size:11px;">IMAGES</button>
                <button id="sd-mode-pdf" style="flex:1;background:#9b59b6;color:#fff;border:none;padding:8px;cursor:pointer;font-weight:bold;border-radius:4px;font-size:11px;">PDF</button>
            </div>
        </div>

        <div style="margin-bottom:10px;background:#2c2c2c;padding:8px;border-radius:4px;display:flex;justify-content:space-between;">
            <div style="text-align:center;">
                <span style="color:#aaa;display:block;margin-bottom:2px;">Start Page</span>
                <input type="number" id="sd-start" value="1" style="width:50px;background:#444;color:#fff;border:1px solid #555;text-align:center;">
            </div>
            <div style="text-align:center;">
                <span style="color:#aaa;display:block;margin-bottom:2px;">Total Pages</span>
                <input type="number" id="sd-end" value="${initialMax}" style="width:50px;background:#444;color:#fff;border:1px solid #555;text-align:center;">
            </div>
            <div style="text-align:center;">
                <span style="color:#aaa;display:block;margin-bottom:2px;">Parallel</span>
                <input type="number" id="sd-parallel" value="${config.concurrency}" min="1" max="12" style="width:50px;background:#444;color:#fff;border:1px solid #555;text-align:center;">
            </div>
        </div>

        <div style="margin-bottom:10px;color:#888;font-size:10px;text-align:center;white-space:nowrap;overflow:hidden;text-overflow:ellipsis;">
            Name: <span style="color:#fff;">${bookName}</span>
        </div>

        <div style="margin-bottom:10px;">
            <div style="display:flex;justify-content:space-between;margin-bottom:3px;">
                <span id="sd-progress-text" style="color:#aaa;font-size:10px;">Ready...</span>
                <span id="sd-progress-percent" style="color:#1abc9c;font-size:10px;font-weight:bold;">0%</span>
            </div>
            <div style="background:#333;height:8px;border-radius:4px;overflow:hidden;">
                <div id="sd-progress-bar" style="background:linear-gradient(90deg, #1abc9c, #16a085);height:100%;width:0%;transition:width 0.3s;"></div>
            </div>
        </div>

        <div id="sd-status" style="margin-bottom:10px;color:#1abc9c;padding:5px;border:1px solid #333;min-height:20px;white-space:nowrap;overflow:hidden;">Ready...</div>

        <button id="sd-go" style="width:100%;background:#1abc9c;color:#fff;border:none;padding:12px;cursor:pointer;font-weight:bold;font-size:14px;border-radius:4px;">START DOWNLOAD</button>
        <div style="font-size:10px;color:#666;text-align:center;margin-top:5px;">Keep PiP window open!</div>
    `;
    document.body.appendChild(panel);

    /* --- CONTENT MODE BUTTON HANDLER --- */
    const contentBtn = document.getElementById('sd-content');
    if (contentBtn && !inContentMode) {
        contentBtn.onclick = goToContentMode;
    }

    /* --- VARIABLES --- */
    let isRunning = false;
    let pageCounter = 1;
    let maxPages = initialMax;
    let downloadMode = 'images'; // 'images' or 'pdf'
    let pdfPages = new Map(); // page number -> raw image bytes
    let doneCount = 0;
    let totalCount = 0;
    let pipActive = false;
    let runId = 0; // guards against an old loop finishing after STOP -> RESUME

    const statusEl = document.getElementById('sd-status');
    const goBtn = document.getElementById('sd-go');
    const startInput = document.getElementById('sd-start');
    const endInput = document.getElementById('sd-end');
    const parallelInput = document.getElementById('sd-parallel');
    const progressBar = document.getElementById('sd-progress-bar');
    const progressText = document.getElementById('sd-progress-text');
    const progressPercent = document.getElementById('sd-progress-percent');
    const modeImagesBtn = document.getElementById('sd-mode-images');
    const modePdfBtn = document.getElementById('sd-mode-pdf');

    /* --- MODE SELECTION --- */
    function setMode(mode) {
        downloadMode = mode;
        if (mode === 'images') {
            modeImagesBtn.style.background = '#3498db';
            modeImagesBtn.style.opacity = '1';
            modePdfBtn.style.background = '#555';
            modePdfBtn.style.opacity = '0.5';
        } else {
            modePdfBtn.style.background = '#9b59b6';
            modePdfBtn.style.opacity = '1';
            modeImagesBtn.style.background = '#555';
            modeImagesBtn.style.opacity = '0.5';
        }
    }

    modeImagesBtn.onclick = () => setMode('images');
    modePdfBtn.onclick = () => setMode('pdf');
    setMode('images'); // Default

    const wait = (ms) => new Promise(r => setTimeout(r, ms));

    const log = (msg, color="#1abc9c") => {
        statusEl.innerText = msg;
        statusEl.style.color = color;
        if(isRunning) document.title = `[${doneCount}/${totalCount}] ${msg}`;
    };

    const updateProgress = () => {
        const percent = totalCount > 0 ? Math.min(100, (doneCount / totalCount) * 100) : 0;
        progressBar.style.width = percent + '%';
        progressPercent.innerText = Math.round(percent) + '%';
        progressText.innerText = `Done ${doneCount} of ${totalCount} (scanning page ${Math.min(pageCounter, maxPages)})`;
    };

    /* --- PIP HACK --- */
    async function activatePiP() {
        if (pipActive) return;
        try {
            const canvas = document.createElement('canvas');
            canvas.width = 300;
            canvas.height = 100;
            const ctx = canvas.getContext('2d');
            setInterval(() => {
                ctx.fillStyle = '#000';
                ctx.fillRect(0,0,300,100);
                ctx.fillStyle = '#1abc9c';
                ctx.font = '20px monospace';
                ctx.fillText(`Done: ${doneCount} / ${totalCount}`, 20, 55);
                ctx.fillStyle = '#fff';
                ctx.font = '12px sans-serif';
                ctx.fillText("Background Active", 20, 80);
            }, 1000);
            const video = document.createElement('video');
            video.srcObject = canvas.captureStream(10);
            video.autoplay = true;
            video.muted = true;       // required for autoplay on iOS
            video.playsInline = true; // otherwise iOS Safari goes fullscreen
            video.onloadedmetadata = async () => {
                try {
                    await video.play();
                    await video.requestPictureInPicture();
                    pipActive = true;
                } catch(e){
                    console.log("PiP activation failed:", e);
                }
            };
        } catch(e){
            console.log("PiP not supported:", e);
        }
    }

    /* --- PAGE URL DISCOVERY --- */
    // Returns the real image URL of a page, or null if it has not been loaded yet.
    // Scribd sometimes keeps the real URL in "orig"/"data-src" before lazy-loading,
    // which lets us skip scrolling entirely.
    function getImageUrl(pageEl) {
        const img = pageEl.querySelector(config.selectors.image);
        if (!img) return null;
        const candidates = [img.getAttribute('orig'), img.getAttribute('data-src'), img.getAttribute('src')];
        for (const url of candidates) {
            if (url && !url.startsWith('data:') && !/loading|blank|spacer|\.gif(\?|$)/i.test(url)) {
                return new URL(url, window.location.href).href;
            }
        }
        return null;
    }

    const PAGE_MISSING = 'missing';

    async function resolvePageUrl(pageNum) {
        const selector = config.selectors.pageContainer(pageNum);
        let pageEl = document.querySelector(selector);

        // Fast path: URL is already in the DOM, no scrolling or waiting needed.
        let url = pageEl && getImageUrl(pageEl);
        if (url) return url;

        // Page container not rendered yet: scroll down until it appears.
        let start = Date.now();
        while (!pageEl && isRunning && Date.now() - start < config.pageTimeout) {
            log(`Page ${pageNum} not found. Scrolling...`, "#f39c12");
            window.scrollBy(0, config.scrollStep);
            await wait(config.pollInterval * 2);
            pageEl = document.querySelector(selector);
        }
        if (!pageEl) return isRunning ? PAGE_MISSING : null;

        // Bring the page into view and poll until the lazy loader sets its URL,
        // instead of always waiting a fixed amount of time.
        pageEl.scrollIntoView({ block: "center", behavior: "auto" });
        start = Date.now();
        while (isRunning && Date.now() - start < config.pageTimeout) {
            url = getImageUrl(pageEl);
            if (url) return url;
            await wait(config.pollInterval);
        }
        return null;
    }

    /* --- DOWNLOAD LOGIC --- */
    async function fetchBytes(url) {
        for (let attempt = 1; ; attempt++) {
            try {
                const response = await fetch(url);
                if (!response.ok) throw new Error(`HTTP ${response.status}`);
                return new Uint8Array(await response.arrayBuffer());
            } catch(e) {
                if (attempt >= config.retryAttempts) throw e;
                await wait(config.retryDelay * attempt);
            }
        }
    }

    function isPng(bytes) {
        return bytes[0] === 0x89 && bytes[1] === 0x50 && bytes[2] === 0x4E && bytes[3] === 0x47;
    }

    function saveBlob(blob, filename) {
        const url = URL.createObjectURL(blob);
        const a = document.createElement('a');
        a.href = url;
        a.download = filename;
        document.body.appendChild(a);
        a.click();
        // Safari needs the object URL alive for a moment after click()
        setTimeout(() => {
            document.body.removeChild(a);
            URL.revokeObjectURL(url);
        }, 1500);
    }

    // Fetches run in parallel, but file saves are serialized with a small gap:
    // Safari silently drops downloads that are triggered at the same instant.
    let saveQueue = Promise.resolve();
    function queueSave(blob, filename) {
        saveQueue = saveQueue.then(() => {
            saveBlob(blob, filename);
            return wait(config.saveGap);
        });
        return saveQueue;
    }

    async function processPage(pageNum, url) {
        try {
            const bytes = await fetchBytes(url);
            if (downloadMode === 'images') {
                const png = isPng(bytes);
                const blob = new Blob([bytes], { type: png ? 'image/png' : 'image/jpeg' });
                await queueSave(blob, `${bookName}_Page_${String(pageNum).padStart(3,'0')}.${png ? 'png' : 'jpg'}`);
            } else {
                pdfPages.set(pageNum, bytes);
            }
            return null;
        } catch(e) {
            console.error(`Page ${pageNum} failed:`, e);
            return (e && e.message) || String(e); // e.g. "Load failed" = blocked by CORS
        }
    }

    async function generatePDF() {
        try {
            if (pdfPages.size === 0) {
                alert("No pages to export!");
                return;
            }
            log(`Generating PDF from ${pdfPages.size} pages...`, "#9b59b6");
            const { jsPDF } = window.jspdf;
            const pageNums = [...pdfPages.keys()].sort((a, b) => a - b);
            // Start with a throwaway page; each real page gets its own size below
            const pdf = new jsPDF({ unit: 'mm' });

            for (let i = 0; i < pageNums.length; i++) {
                const bytes = pdfPages.get(pageNums[i]);
                const format = isPng(bytes) ? 'PNG' : 'JPEG';
                // Passing raw bytes lets jsPDF embed the JPEG as-is (no canvas re-encode)
                const props = pdf.getImageProperties(bytes);
                const w = props.width * 0.264583; // px to mm
                const h = props.height * 0.264583;
                const orientation = w > h ? 'landscape' : 'portrait';

                pdf.addPage([w, h], orientation);
                // Unique alias skips jsPDF's hashing of every image
                pdf.addImage(bytes, format, 0, 0, w, h, `page${pageNums[i]}`, 'NONE');

                if (i % 10 === 9) {
                    log(`Building PDF: ${i+1}/${pageNums.length}`, "#9b59b6");
                    await wait(0); // let the UI repaint
                }
            }

            pdf.deletePage(1);
            saveBlob(pdf.output('blob'), `${bookName}.pdf`);
            log("PDF Downloaded!", "#27ae60");
            alert("PDF generated successfully!");
        } catch(e) {
            console.error("PDF generation error:", e);
            alert("PDF generation failed: " + e.message);
        }
    }

    /* --- MAIN LOOP --- */
    // The scanner walks pages in order and only discovers image URLs;
    // fetching happens in a pool of parallel workers so the network
    // never sits idle while we wait for the next page to lazy-load.
    async function startLoop() {
        const concurrency = Math.max(1, Math.min(12, parseInt(parallelInput.value) || config.concurrency));
        const inflight = new Set();
        const failed = [];
        let firstError = '';
        let stoppedEarly = false;
        const myRun = ++runId;

        while (isRunning && myRun === runId && pageCounter <= maxPages) {
            const pageNum = pageCounter;
            updateProgress();
            log(`Scanning Page ${pageNum}...`, "#fff");

            const url = await resolvePageUrl(pageNum);
            if (!isRunning || myRun !== runId) break;
            if (url === PAGE_MISSING) {
                log(`Page ${pageNum} missing. Stopping.`, "#e74c3c");
                stoppedEarly = true;
                break;
            }

            if (!url) {
                log(`Skipping Page ${pageNum} (No Image)`, "#e67e22");
                failed.push(pageNum);
                firstError = firstError || 'image not found on page';
                doneCount++;
            } else {
                while (inflight.size >= concurrency) await Promise.race(inflight);
                const task = processPage(pageNum, url).then(err => {
                    if (err) {
                        failed.push(pageNum);
                        firstError = firstError || err;
                        log(`Page ${pageNum} failed: ${err}`, "#e74c3c");
                    }
                    doneCount++;
                    updateProgress();
                }).finally(() => inflight.delete(task));
                inflight.add(task);
            }

            pageCounter++;
            startInput.value = pageCounter;
        }

        if (inflight.size > 0) log(`Finishing ${inflight.size} downloads...`, "#fff");
        await Promise.all(inflight);
        await saveQueue;

        if (myRun !== runId) return; // a newer run took over
        if (!isRunning) {
            log(`Paused at page ${pageCounter}.`, "#f39c12");
            return;
        }
        isRunning = false;

        if (downloadMode === 'pdf') {
            await generatePDF();
        }

        failed.sort((a, b) => a - b);
        const summary = failed.length ? `\nFailed pages: ${failed.join(', ')}\nReason: ${firstError}` : '';
        log(failed.length ? `Done, ${failed.length} failed` : "Done!", failed.length ? "#e67e22" : "#27ae60");
        alert((stoppedEarly ? "Stopped early." : "Download Complete!") + summary);
        goBtn.innerText = "COMPLETED";
        goBtn.style.background = "#27ae60";
        if (!stoppedEarly) {
            progressBar.style.width = "100%";
            progressPercent.innerText = "100%";
        }
    }

    /* --- START HANDLER --- */
    goBtn.onclick = async () => {
        if (isRunning) {
            isRunning = false;
            goBtn.innerText = "RESUME";
            goBtn.style.background = "#1abc9c";
            return;
        }

        // Page images are only reachable in the embed viewer
        if (!isContentMode()) {
            if (confirm("Downloading only works in CONTENT MODE.\nSwitch to content mode now?")) goToContentMode();
            return;
        }

        const resuming = goBtn.innerText === "RESUME";
        activatePiP();

        isRunning = true;
        pageCounter = parseInt(startInput.value);
        maxPages = parseInt(endInput.value);
        if (!resuming) {
            pdfPages = new Map();
            doneCount = 0;
            totalCount = maxPages - pageCounter + 1;
        }

        goBtn.innerText = "STOP";
        goBtn.style.background = "#e74c3c";
        log("Started!", "#fff");
        updateProgress();

        startLoop();
    };
})();
